### `scripts/copy-stockfish.mjs`
```js
// Copies the real Stockfish.js lite single-threaded WASM build from node_modules into public/engine
// and writes a manifest listing only what is actually bundled.
import { cpSync, existsSync, mkdirSync, readdirSync, readFileSync, rmSync, writeFileSync } from "node:fs";
import { join } from "node:path";

const pkgDir = join("node_modules", "stockfish");
if (!existsSync(pkgDir)) {
  console.error("The 'stockfish' package is not installed. Run: npm install");
  process.exit(1);
}
const pkg = JSON.parse(readFileSync(join(pkgDir, "package.json"), "utf8"));

let found = null;
for (const dir of ["bin", "src", "."].map((d) => join(pkgDir, d)).filter(existsSync)) {
  const files = readdirSync(dir);
  const js = files
    .filter((f) => /lite-single.*\.js$/.test(f) && files.includes(f.replace(/\.js$/, ".wasm")))
    .sort()[0];
  if (js) { found = { dir, js }; break; }
}
if (!found) {
  console.error("No lite-single JS+WASM pair found in node_modules/stockfish (looked in bin/, src/, root).");
  process.exit(1);
}

const out = join("public", "engine");
rmSync(out, { recursive: true, force: true });
mkdirSync(out, { recursive: true });
const wasm = found.js.replace(/\.js$/, ".wasm");
cpSync(join(found.dir, found.js), join(out, found.js));
cpSync(join(found.dir, wasm), join(out, wasm));
for (const name of ["COPYING.txt", "Copying.txt", "COPYING", "LICENSE", "LICENSE.txt"]) {
  if (existsSync(join(pkgDir, name))) { cpSync(join(pkgDir, name), join(out, "COPYING.txt")); break; }
}

const fromName = /stockfish[a-z-]*?-?(\d+(?:\.\d+)?)/i.exec(found.js)?.[1];
const version = fromName ?? pkg.version;
writeFileSync(
  join(out, "manifest.json"),
  JSON.stringify([{
    id: "stockfish-lite",
    name: `Stockfish ${version} Lite (single-threaded WASM)`,
    version, js: found.js, wasm, license: "GPL-3.0-or-later",
    source: "https://github.com/official-stockfish/Stockfish",
    build: "https://github.com/nmrugg/stockfish.js",
  }], null, 2),
);
console.log(`Bundled ${found.js} (+ .wasm) -> public/engine`);
```

```ts
import { Chess } from "chess.js";
import type { Move } from "chess.js";

export const START_FEN = "rnbqkbnr/pppppppp/8/8/8/8/PPPPPPPP/RNBQKBNR w KQkq - 0 1";
export type Side = "w" | "b";
export type Result = "1-0" | "0-1" | "1/2-1/2";
export type Termination =
  | "checkmate" | "stalemate" | "insufficient_material" | "fivefold_repetition" | "seventyfive_move"
  | "threefold_repetition" | "fifty_move" | "timeout" | "resignation" | "agreement";
export interface Outcome { result: Result; termination: Termination }

export const TIME_CONTROLS = [
  { id: "1+0", initial: 60_000, increment: 0, kind: "Bullet" },
  { id: "2+1", initial: 120_000, increment: 1_000, kind: "Bullet" },
  { id: "3+2", initial: 180_000, increment: 2_000, kind: "Blitz" },
  { id: "5+0", initial: 300_000, increment: 0, kind: "Blitz" },
  { id: "10+0", initial: 600_000, increment: 0, kind: "Rapid" },
  { id: "10+5", initial: 600_000, increment: 5_000, kind: "Rapid" },
] as const;
export const findTimeControl = (id: unknown) => TIME_CONTROLS.find((t) => t.id === id);

export const AI_LEVELS = [
  { id: 1, label: "Beginner", elo: 1320, skill: 1 },
  { id: 2, label: "Casual", elo: 1600, skill: 5 },
  { id: 3, label: "Club", elo: 1900, skill: 9 },
  { id: 4, label: "Strong", elo: 2200, skill: 13 },
  { id: 5, label: "Expert", elo: 2600, skill: 17 },
  { id: 6, label: "Maximum", elo: null, skill: 20 },
] as const;

export const UCI_RE = /^[a-h][1-8][a-h][1-8][qrbn]?$/;
export const positionKey = (fen: string) => fen.split(" ").slice(0, 4).join(" ");
const halfmove = (c: Chess) => Number(c.fen().split(" ")[4]);

export function applyUci(chess: Chess, uci: string): Move {
  if (!UCI_RE.test(uci)) throw new Error("Malformed move");
  return chess.move({ from: uci.slice(0, 2), to: uci.slice(2, 4), promotion: uci[4] });
}

/** Replays a game from the start; keys[i] is the repetition key after ply i (keys[0] = start). */
export function replay(ucis: string[]) {
  const chess = new Chess();
  const keys = [positionKey(chess.fen())];
  for (const u of ucis) { applyUci(chess, u); keys.push(positionKey(chess.fen())); }
  return { chess, keys };
}

const repCount = (keys: string[]) => keys.filter((k) => k === keys[keys.length - 1]).length;
const draw = (termination: Termination): Outcome => ({ result: "1/2-1/2", termination });

/** Endings that happen automatically (no claim needed). Checkmate always takes precedence. */
export function automaticOutcome(chess: Chess, keys: string[] = [positionKey(chess.fen())]): Outcome | null {
  if (chess.isCheckmate()) {
    return { result: chess.turn() === "w" ? "0-1" : "1-0", termination: "checkmate" };
  }
  if (chess.isStalemate()) return draw("stalemate");
  if (chess.isInsufficientMaterial()) return draw("insufficient_material");
  if (repCount(keys) >= 5) return draw("fivefold_repetition");
  if (halfmove(chess) >= 150) return draw("seventyfive_move");
  return null;
}

/** Draws a player may claim (threefold repetition, 50-move rule). Not automatic. */
export function claimableDraw(chess: Chess, keys: string[] = [positionKey(chess.fen())]): Outcome | null {
  if (repCount(keys) >= 3) return draw("threefold_repetition");
  if (halfmove(chess) >= 100) return draw("fifty_move");
  return null;
}

/** Can `side` still checkmate with the material on the board (so a flag fall is a real win)? */
export function hasMatingMaterial(chess: Chess, side: Side): boolean {
  let strong = false, knights = 0, light = 0, dark = 0, oppPieces = 0, oppNP = 0;
  chess.board().forEach((row, r) => row.forEach((cell, c) => {
    if (!cell) return;
    if (cell.color === side) {
      if (cell.type === "p" || cell.type === "r" || cell.type === "q") strong = true;
      else if (cell.type === "n") knights++;
      else if (cell.type === "b") ((r + c) % 2 === 0 ? light++ : dark++);
    } else if (cell.type !== "k") {
      oppPieces++;
      if (cell.type === "n" || cell.type === "p") oppNP++;
    }
  }));
  if (strong || (light > 0 && dark > 0) || knights >= 2 || (knights >= 1 && light + dark >= 1)) return true;
  if (knights === 1) return oppPieces > 0;
  if (light + dark > 0) return oppNP > 0;
  return false;
}

export function timeoutOutcome(chess: Chess, flagged: Side): Outcome {
  const winner: Side = flagged === "w" ? "b" : "w";
  if (!hasMatingMaterial(chess, winner)) return draw("timeout");
  return { result: winner === "w" ? "1-0" : "0-1", termination: "timeout" };
}

export interface ClockState {
  white_ms: number; black_ms: number; increment_ms: number;
  turn: Side; turn_started_at: string | null; ply: number; status: string;
}
export const clockRunning = (g: Pick<ClockState, "status" | "ply" | "turn_started_at">) =>
  g.status === "active" && g.ply >= 2 && !!g.turn_started_at;

/** Visible remaining time for both sides at `nowMs` (client display; nothing is written per second). */
export function remainingAt(g: ClockState, nowMs: number) {
  let w = g.white_ms, b = g.black_ms;
  if (clockRunning(g)) {
    const el = Math.max(0, nowMs - Date.parse(g.turn_started_at!));
    if (g.turn === "w") w = Math.max(0, w - el); else b = Math.max(0, b - el);
  }
  return { w, b };
}

/** Server-side clock update for a move made now. Clocks start after each side's first move; no increment on those. */
export function clockAfterMove(g: ClockState, nowMs: number) {
  const running = g.ply >= 2 && !!g.turn_started_at;
  const elapsed = running ? Math.max(0, nowMs - Date.parse(g.turn_started_at!)) : 0;
  const key = g.turn === "w" ? "white_ms" : "black_ms";
  const left = g[key] - elapsed;
  const out = { white_ms: g.white_ms, black_ms: g.black_ms, flagged: false };
  if (left <= 0) { out[key] = 0; out.flagged = true; return out; }
  out[key] = left + (running ? g.increment_ms : 0);
  return out;
}

export function buildPgn(
  ucis: string[],
  m: { white: string; black: string; result: Result | "*"; tc: string; initial: number; inc: number; date: string; mode: string },
): string {
  const c = new Chess();
  for (const u of ucis) applyUci(c, u);
  c.setHeader("Event", m.mode === "ai" ? "Computer game" : "Online game");
  c.setHeader("Site", "Chess");
  c.setHeader("Date", m.date.slice(0, 10).replace(/-/g, "."));
  c.setHeader("White", m.white);
  c.setHeader("Black", m.black);
  c.setHeader("Result", m.result);
  c.setHeader("TimeControl", `${m.initial / 1000}+${m.inc / 1000}`);
  return c.pgn();
}
```

```ts
// Single trusted endpoint. Authenticates the caller, validates everything with chess.js, and commits via
// compare-and-swap on `revision`. The browser never writes game state directly.
import { createClient } from "@supabase/supabase-js";
import {
  AI_LEVELS, applyUci, automaticOutcome, buildPgn, claimableDraw, clockAfterMove, findTimeControl,
  positionKey, replay, timeoutOutcome, UCI_RE, type Outcome, type Side,
} from "./rules.ts";

type Row = Record<string, any>;

const admin = createClient(
  Deno.env.get("SUPABASE_URL")!,
  Deno.env.get("SERVICE_KEY") ?? Deno.env.get("SUPABASE_SERVICE_ROLE_KEY")!,
  { auth: { persistSession: false, autoRefreshToken: false } },
);

const cors = {
  "Access-Control-Allow-Origin": "*",
  "Access-Control-Allow-Headers": "authorization, x-client-info, apikey, content-type",
  "Access-Control-Allow-Methods": "POST, OPTIONS",
};
const json = (body: unknown, status = 200) =>
  new Response(JSON.stringify(body), { status, headers: { ...cors, "Content-Type": "application/json" } });

class HttpError extends Error {
  constructor(public status: number, message: string, public code?: string) { super(message); }
}

const displayName = (u: any) =>
  String(u.user_metadata?.full_name ?? u.user_metadata?.name ?? u.email?.split("@")[0] ?? "Player").slice(0, 40);
const avatarOf = (u: any) => {
  const a = u.user_metadata?.avatar_url ?? u.user_metadata?.picture;
  return typeof a === "string" && a.startsWith("https://") ? a.slice(0, 500) : null;
};

async function getGame(id: unknown): Promise<Row> {
  if (typeof id !== "string" || !/^[0-9a-f-]{36}$/i.test(id)) throw new HttpError(400, "Bad game id");
  const { data, error } = await admin.from("games").select("*").eq("id", id).maybeSingle();
  if (error) throw error;
  if (!data) throw new HttpError(404, "Game not found");
  return data;
}
function mySide(g: Row, uid: string): Side {
  if (g.white_id === uid) return "w";
  if (g.black_id === uid) return "b";
  throw new HttpError(403, "Not your game");
}
async function fetchMoves(id: string): Promise<Row[]> {
  const out: Row[] = [];
  for (let from = 0; ; from += 1000) {
    const { data, error } = await admin.from("game_moves").select("*").eq("game_id", id).order("ply").range(from, from + 999);
    if (error) throw error;
    out.push(...data);
    if (data.length < 1000) break;
  }
  return out;
}
async function ucisOf(g: Row): Promise<string[]> {
  const ucis = (await fetchMoves(g.id)).map((m) => m.uci as string);
  if (ucis.length !== g.ply) throw new HttpError(409, "Game is out of sync, try again", "stale");
  return ucis;
}

function finishPatch(g: Row, ucis: string[], out: Outcome, extra: Row = {}): Row {
  return {
    status: "finished", result: out.result, termination: out.termination, ended_at: new Date().toISOString(),
    draw_offer_by: null,
    pgn: buildPgn(ucis, {
      white: g.white_name ?? "White", black: g.black_name ?? "Black", result: out.result, tc: g.tc,
      initial: g.initial_ms, inc: g.increment_ms, date: g.created_at, mode: g.mode,
    }),
    ...extra,
  };
}

async function casUpdate(g: Row, patch: Row): Promise<Row | null> {
  const { data, error } = await admin.from("games")
    .update({ ...patch, revision: g.revision + 1, updated_at: new Date().toISOString() })
    .eq("id", g.id).eq("revision", g.revision).eq("status", "active")
    .select().maybeSingle();
  if (error) throw error;
  return data;
}
async function mutate(gameId: unknown, uid: string, fn: (g: Row, me: Side) => Promise<Row> | Row): Promise<Row> {
  for (let i = 0; i < 4; i++) {
    const g = await getGame(gameId);
    const me = mySide(g, uid);
    if (g.status !== "active") throw new HttpError(409, "Game is not active", "not_active");
    const row = await casUpdate(g, await fn(g, me));
    if (row) return row;
  }
  throw new HttpError(409, "The game changed, please retry", "stale");
}

async function doMove(user: any, b: Row): Promise<Row> {
  const g = await getGame(b.gameId);
  const me = mySide(g, user.id);
  if (g.status !== "active") throw new HttpError(409, "Game is over", "not_active");
  const asAi = b.ai === true;
  if (asAi && g.mode !== "ai") throw new HttpError(400, "Not an engine game");
  const mover: Side = asAi ? g.ai_color : me;
  if (!asAi && g.mode === "ai" && mover === g.ai_color) throw new HttpError(403, "Not your side");
  if (g.turn !== mover) throw new HttpError(409, "It is not your turn", "not_your_turn");
  if (b.expectedRevision !== g.revision) throw new HttpError(409, "Position changed, reloading", "stale");

  const ucis = await ucisOf(g);
  const { chess, keys } = replay(ucis);
  const clk = clockAfterMove(g as any, Date.now());
  if (clk.flagged) {
    const out = timeoutOutcome(chess, mover);
    const row = await casUpdate(g, finishPatch(g, ucis, out, { [mover === "w" ? "white_ms" : "black_ms"]: 0 }));
    if (!row) throw new HttpError(409, "Position changed, reloading", "stale");
    return { game: row, flagged: true };
  }
  if (typeof b.uci !== "string" || !UCI_RE.test(b.uci)) throw new HttpError(400, "Malformed move");
  let mv;
  try { mv = applyUci(chess, b.uci); } catch { throw new HttpError(400, "Illegal move", "illegal"); }
  const uci = `${mv.from}${mv.to}${mv.promotion ?? ""}`;
  keys.push(positionKey(chess.fen()));
  const out = automaticOutcome(chess, keys);
  const pgn = buildPgn([...ucis, uci], {
    white: g.white_name ?? "White", black: g.black_name ?? "Black", result: out?.result ?? "*", tc: g.tc,
    initial: g.initial_ms, inc: g.increment_ms, date: g.created_at, mode: g.mode,
  });
  const fen = chess.fen();
  const { data, error } = await admin.rpc("commit_move", {
    p_game: g.id, p_expected_rev: g.revision, p_ply: g.ply + 1, p_player: asAi ? null : user.id,
    p_san: mv.san, p_uci: uci, p_fen: fen, p_pgn: pgn, p_white_ms: clk.white_ms, p_black_ms: clk.black_ms,
    p_turn: chess.turn(), p_result: out?.result ?? null, p_termination: out?.termination ?? null,
  });
  if (error) throw error;
  const row = Array.isArray(data) ? data[0] : data;
  if (!row) throw new HttpError(409, "Position changed, reloading", "stale");
  return {
    game: row,
    move: { game_id: g.id, ply: g.ply + 1, player_id: asAi ? null : user.id, san: mv.san, uci, fen,
      white_ms: clk.white_ms, black_ms: clk.black_ms },
  };
}

async function handle(user: any, b: Row): Promise<Row> {
  switch (b.action) {
    case "queue": {
      const tc = findTimeControl(b.tc);
      if (!tc) throw new HttpError(400, "Unknown time control");
      const { data, error } = await admin.rpc("queue_game", {
        p_user: user.id, p_name: displayName(user), p_avatar: avatarOf(user),
        p_tc: tc.id, p_initial: tc.initial, p_inc: tc.increment,
      });
      if (error) {
        if (error.code === "23505") throw new HttpError(409, "You are already in the queue", "duplicate");
        throw error;
      }
      return { game: Array.isArray(data) ? data[0] : data };
    }
    case "touch_queue": {
      const { data, error } = await admin.from("games").update({ updated_at: new Date().toISOString() })
        .eq("id", b.gameId).eq("white_id", user.id).eq("status", "waiting").select().maybeSingle();
      if (error) throw error;
      return { game: data ?? await getGame(b.gameId) };
    }
    case "cancel_queue": {
      const { data, error } = await admin.from("games").delete()
        .eq("id", b.gameId).eq("white_id", user.id).eq("status", "waiting").eq("mode", "online").select().maybeSingle();
      if (error) throw error;
      if (data) return { game: { ...data, status: "aborted" } };
      const g = await getGame(b.gameId);
      mySide(g, user.id);
      return { game: g };
    }
    case "new_ai": {
      const tc = findTimeControl(b.tc);
      const level = AI_LEVELS.find((l) => l.id === b.level);
      const human: Side | null = b.humanColor === "w" || b.humanColor === "b" ? b.humanColor : null;
      if (!tc || !level || !human) throw new HttpError(400, "Invalid game settings");
      if (typeof b.engine !== "string" || !/^[a-z0-9.-]{1,64}$/.test(b.engine)) throw new HttpError(400, "Invalid engine");
      const aiName = String(b.aiName ?? "Stockfish").slice(0, 60);
      const now = new Date().toISOString();
      const { data, error } = await admin.from("games").insert({
        mode: "ai", status: "active", ai_engine: b.engine, ai_level: level.id, ai_color: human === "w" ? "b" : "w",
        white_id: human === "w" ? user.id : null, black_id: human === "b" ? user.id : null,
        white_name: human === "w" ? displayName(user) : aiName, black_name: human === "b" ? displayName(user) : aiName,
        white_avatar: human === "w" ? avatarOf(user) : null, black_avatar: human === "b" ? avatarOf(user) : null,
        tc: tc.id, initial_ms: tc.initial, increment_ms: tc.increment, white_ms: tc.initial, black_ms: tc.initial,
        started_at: now, turn_started_at: now,
      }).select().single();
      if (error) throw error;
      return { game: data };
    }
    case "sync": {
      const g = await getGame(b.gameId);
      mySide(g, user.id);
      return { game: g, moves: await fetchMoves(g.id) };
    }
    case "move": return await doMove(user, b);
    case "resign":
      return { game: await mutate(b.gameId, user.id, async (g, me) =>
        finishPatch(g, await ucisOf(g), { result: me === "w" ? "0-1" : "1-0", termination: "resignation" })) };
    case "offer_draw":
      return { game: await mutate(b.gameId, user.id, async (g) => {
        if (g.mode !== "online") throw new HttpError(400, "Draw offers are for online games");
        if (g.draw_offer_by && g.draw_offer_by !== user.id) {
          return finishPatch(g, await ucisOf(g), { result: "1/2-1/2", termination: "agreement" });
        }
        return { draw_offer_by: user.id };
      }) };
    case "accept_draw":
      return { game: await mutate(b.gameId, user.id, async (g) => {
        if (!g.draw_offer_by || g.draw_offer_by === user.id) throw new HttpError(409, "There is no draw offer to accept", "no_offer");
        return finishPatch(g, await ucisOf(g), { result: "1/2-1/2", termination: "agreement" });
      }) };
    case "decline_draw":
      return { game: await mutate(b.gameId, user.id, (g) => {
        if (!g.draw_offer_by || g.draw_offer_by === user.id) throw new HttpError(409, "There is no draw offer to decline", "no_offer");
        return { draw_offer_by: null };
      }) };
    case "claim_draw":
      return { game: await mutate(b.gameId, user.id, async (g) => {
        const ucis = await ucisOf(g);
        const { chess, keys } = replay(ucis);
        const claim = claimableDraw(chess, keys);
        if (!claim) throw new HttpError(409, "No draw can be claimed in this position", "not_claimable");
        return finishPatch(g, ucis, claim);
      }) };
    case "timeout":
      return { game: await mutate(b.gameId, user.id, async (g) => {
        const key = g.turn === "w" ? "white_ms" : "black_ms";
        const elapsed = g.ply >= 2 && g.turn_started_at ? Date.now() - Date.parse(g.turn_started_at) : 0;
        if (g.ply < 2 || g[key] - elapsed > 0) throw new HttpError(409, "The clock has not expired", "not_flagged");
        const ucis = await ucisOf(g);
        return finishPatch(g, ucis, timeoutOutcome(replay(ucis).chess, g.turn), { [key]: 0 });
      }) };
    default:
      throw new HttpError(400, "Unknown action");
  }
}

Deno.serve(async (req) => {
  if (req.method === "OPTIONS") return new Response("ok", { headers: cors });
  try {
    if (req.method !== "POST") throw new HttpError(405, "POST only");
    const token = (req.headers.get("Authorization") ?? "").replace(/^Bearer\s+/i, "");
    const { data, error } = await admin.auth.getUser(token);
    if (error || !data.user) throw new HttpError(401, "Please sign in again", "unauthenticated");
    const body = await req.json().catch(() => { throw new HttpError(400, "Invalid JSON"); });
    const out = await handle(data.user, body ?? {});
    return json({ ...out, serverNow: Date.now() });
  } catch (e) {
    if (e instanceof HttpError) return json({ error: e.message, code: e.code }, e.status);
    console.error(e);
    return json({ error: "Server error" }, 500);
  }
```

### `src/lib/engine.ts`
```ts
// Real Stockfish (UCI) running inside a dedicated Web Worker. Nothing here is simulated.
import { AI_LEVELS } from "../../supabase/functions/game-action/rules.ts";

export interface EngineEntry {
  id: string; name: string; version: string; js: string; wasm: string; license: string; source: string; build: string;
}

export async function loadManifest(): Promise<EngineEntry[]> {
  const r = await fetch("/engine/manifest.json", { cache: "no-cache" });
  if (!r.ok) throw new Error("No engine is bundled in this build (missing /engine/manifest.json).");
  const j = await r.json();
  if (!Array.isArray(j) || j.length === 0) throw new Error("The engine manifest is empty.");
  return j as EngineEntry[];
}

type Listener = (line: string) => void;

export class UciEngine {
  private listeners = new Set<Listener>();
  private opts = new Map<string, { min?: number; max?: number }>();
  private searching = false;
  private cancelled = false;

  private constructor(private worker: Worker) {
    worker.onmessage = (e: MessageEvent) => {
      const raw = typeof e.data === "string" ? e.data : String((e.data as { data?: unknown })?.data ?? e.data);
      for (const line of raw.split("\n")) {
        const t = line.trim();
        if (t) this.listeners.forEach((l) => l(t));
      }
    };
  }

  private send(cmd: string) { this.worker.postMessage(cmd); }

  private waitFor(pred: (l: string) => boolean, ms: number): Promise<string> {
    return new Promise((resolve, reject) => {
      const timer = setTimeout(() => { this.listeners.delete(fn); reject(new Error("The engine stopped responding.")); }, ms);
      const fn: Listener = (l) => { if (pred(l)) { clearTimeout(timer); this.listeners.delete(fn); resolve(l); } };
      this.listeners.add(fn);
    });
  }

  static async load(entry: EngineEntry): Promise<UciEngine> {
    const worker = new Worker(`/engine/${entry.js}`); // classic worker; the .wasm sits beside it
    const eng = new UciEngine(worker);
    const failed = new Promise<never>((_, rej) => { worker.onerror = () => rej(new Error("Could not start the engine worker.")); });
    const collect: Listener = (l) => {
      const m = /^option name (.+?) type (\w+)(.*)$/.exec(l);
      if (!m) return;
      const min = /min (-?\d+)/.exec(m[3]);
      const max = /max (-?\d+)/.exec(m[3]);
      eng.opts.set(m[1], { min: min ? +min[1] : undefined, max: max ? +max[1] : undefined });
    };
    try {
      eng.listeners.add(collect);
      const ok = eng.waitFor((l) => l === "uciok", 30_000);
      eng.send("uci");
      await Promise.race([ok, failed]);
      eng.listeners.delete(collect);
      await eng.ready();
      return eng;
    } catch (e) {
      worker.terminate();
      throw e;
    }
  }

  async ready() {
    const p = this.waitFor((l) => l === "readyok", 15_000);
    this.send("isready");
    await p;
  }

  /** Uses UCI_LimitStrength/UCI_Elo when the engine reports them, otherwise Skill Level. */
  async configure(levelId: number) {
    const lv = AI_LEVELS.find((l) => l.id === levelId) ?? AI_LEVELS[2];
    const hasLimit = this.opts.has("UCI_LimitStrength");
    if (lv.elo && hasLimit && this.opts.has("UCI_Elo")) {
      const o = this.opts.get("UCI_Elo")!;
      const elo = Math.min(o.max ?? lv.elo, Math.max(o.min ?? lv.elo, lv.elo));
      this.send("setoption name UCI_LimitStrength value true");
      this.send(`setoption name UCI_Elo value ${elo}`);
    } else {
      if (hasLimit) this.send("setoption name UCI_LimitStrength value false");
      if (this.opts.has("Skill Level")) this.send(`setoption name Skill Level value ${lv.skill}`);
    }
    this.send("ucinewgame");
    await this.ready();
  }

  async bestMove(fen: string, movetimeMs: number): Promise<string | null> {
    if (this.searching) throw new Error("Engine is busy");
    this.searching = true;
    this.cancelled = false;
    try {
      this.send(`position fen ${fen}`);
      const p = this.waitFor((l) => l.startsWith("bestmove"), movetimeMs + 20_000);
      this.send(`go movetime ${Math.round(movetimeMs)}`);
      const line = await p;
      if (this.cancelled) return null;
      const m = /^bestmove (\S+)/.exec(line);
      return m && m[1] !== "(none)" && m[1] !== "0000" ? m[1] : null;
    } finally {
      this.searching = false;
    }
  }

  cancel() { if (this.searching) { this.cancelled = true; this.send("stop"); } }

  dispose() {
    this.listeners.clear();
    try { this.send("quit"); } catch { /* worker already gone */ }
    this.worker.terminate();
  }
}
```

```ts
import { useCallback, useEffect, useState } from "react";
import { act, type ActionResult } from "../lib/api";
import { errMsg } from "../lib/format";
import { supabase } from "../lib/supabase";
import type { Game, MoveRow } from "../lib/types";

export function useOnline() {
  const [on, setOn] = useState(navigator.onLine);
  useEffect(() => {
    const a = () => setOn(true), b = () => setOn(false);
    addEventListener("online", a); addEventListener("offline", b);
    return () => { removeEventListener("online", a); removeEventListener("offline", b); };
  }, []);
  return on;
}

const pick = (cur: Game | null, next: Game) => (!cur || next.revision >= cur.revision ? next : cur);
export function mergeMoves(a: MoveRow[], b: MoveRow[]): MoveRow[] {
  const m = new Map<number, MoveRow>();
  for (const x of [...a, ...b]) m.set(x.ply, x);
  return [...m.values()].sort((x, y) => x.ply - y.ply);
}

/** Authoritative game + full move history, kept live through Realtime and re-synced on reconnect. */
export function useGame(id: string) {
  const [game, setGame] = useState<Game | null>(null);
  const [moves, setMoves] = useState<MoveRow[]>([]);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState<string | null>(null);
  const [live, setLive] = useState<"connecting" | "live" | "offline">("connecting");
  const online = useOnline();

  const sync = useCallback(async () => {
    try {
      const r = await act({ action: "sync", gameId: id });
      setGame((g) => pick(g, r.game));
      setMoves((ms) => mergeMoves(ms, r.moves ?? []));
      setError(null);
    } catch (e) {
      setError(errMsg(e));
    } finally {
      setLoading(false);
    }
  }, [id]);

  const apply = useCallback((r: ActionResult) => {
    setGame((g) => pick(g, r.game));
    if (r.move) setMoves((ms) => mergeMoves(ms, [r.move!]));
  }, []);

  useEffect(() => { if (online) sync(); }, [online, sync]);

  useEffect(() => {
    const onVis = () => { if (document.visibilityState === "visible") sync(); };
    document.addEventListener("visibilitychange", onVis);
    return () => document.removeEventListener("visibilitychange", onVis);
  }, [sync]);

  useEffect(() => {
    const ch = supabase.channel(`game:${id}`)
      .on("postgres_changes", { event: "UPDATE", schema: "public", table: "games", filter: `id=eq.${id}` },
        (p) => setGame((g) => pick(g, p.new as unknown as Game)))
      .on("postgres_changes", { event: "INSERT", schema: "public", table: "game_moves", filter: `game_id=eq.${id}` },
        (p) => setMoves((ms) => mergeMoves(ms, [p.new as unknown as MoveRow])))
      .subscribe((status) => {
        if (status === "SUBSCRIBED") { setLive("live"); sync(); }
        else if (status === "CHANNEL_ERROR" || status === "TIMED_OUT" || status === "CLOSED") setLive("offline");
      });
    return () => { supabase.removeChannel(ch); };
  }, [id, sync]);

  // Safety net: poll lightly while a game is active (faster if Realtime is down).
  useEffect(() => {
    if (game?.status !== "active") return;
    const t = setInterval(() => { if (document.visibilityState === "visible") sync(); }, live === "live" ? 30_000 : 8_000);
    return () => clearInterval(t);
  }, [game?.status, live, sync]);

  // If the move list is behind the game row, reload the full history.
  useEffect(() => {
    if (game && moves.length < game.ply) { const t = setTimeout(sync, 700); return () => clearTimeout(t); }
  }, [game?.ply, moves.length, sync]);

  return { game, moves, loading, error, live, online, sync, apply };
}
```

```ts
import { useCallback, useEffect, useRef, useState } from "react";
import { serverNow } from "../lib/api";
import { loadManifest, UciEngine } from "../lib/engine";
import { errMsg } from "../lib/format";
import type { Game } from "../lib/types";

type Status = "off" | "loading" | "ready" | "thinking" | "error";

/** Loads the bundled engine, and plays the engine side by asking it for real UCI moves. */
export function useAiPlayer(game: Game, submit: (uci: string) => Promise<boolean>) {
  const [state, setState] = useState<{ status: Status; error?: string }>({ status: "off" });
  const [paused, setPaused] = useState(false);
  const [attempt, setAttempt] = useState(0);
  const [kick, setKick] = useState(0);
  const [ready, setReady] = useState(false);
  const engineRef = useRef<UciEngine | null>(null);
  const submitRef = useRef(submit);
  submitRef.current = submit;
  const enabled = game.mode === "ai" && game.status === "active";

  useEffect(() => {
    if (!enabled) return;
    let dead = false;
    setReady(false);
    setState({ status: "loading" });
    (async () => {
      try {
        const entry = (await loadManifest()).find((e) => e.id === game.ai_engine);
        if (!entry) throw new Error(`Engine "${game.ai_engine}" is not bundled in this build.`);
        const eng = await UciEngine.load(entry);
        if (dead) { eng.dispose(); return; }
        await eng.configure(game.ai_level ?? 3);
        if (dead) { eng.dispose(); return; }
        engineRef.current = eng;
        setReady(true);
        setState({ status: "ready" });
      } catch (e) {
        if (!dead) setState({ status: "error", error: errMsg(e) });
      }
    })();
    return () => { dead = true; engineRef.current?.dispose(); engineRef.current = null; setReady(false); };
  }, [enabled, game.ai_engine, game.ai_level, attempt]);

  useEffect(() => {
    const eng = engineRef.current;
    if (!enabled || !ready || !eng || paused) return;
    if (game.turn !== game.ai_color) { setState((s) => (s.status === "thinking" ? { status: "ready" } : s)); return; }
    let stale = false;
    const rem = remainingAt(game, serverNow())[game.ai_color!];
    const movetime = Math.min(2000, Math.max(150, rem / 40 + game.increment_ms * 0.6));
    setState({ status: "thinking" });
    eng.bestMove(game.fen, movetime)
      .then(async (uci) => {
        if (stale) return;
        if (!uci) throw new Error("The engine returned no move.");
        const ok = await submitRef.current(uci);
        if (!ok && !stale) setTimeout(() => setKick((k) => k + 1), 1200);
      })
      .catch((e) => { if (!stale) setState({ status: "error", error: errMsg(e) }); });
    return () => { stale = true; eng.cancel(); };
  }, [enabled, ready, paused, game.revision, game.turn, game.status, kick]);

  return {
    ...state, paused,
    stop: useCallback(() => setPaused(true), []),
    resume: useCallback(() => setPaused(false), []),
    retry: useCallback(() => { setAttempt((a) => a + 1); setKick((k) => k + 1); }, []),
  };
}
```

```tsx
function Play({ game, moves, live, online, sync, apply, userId, backTo }: PlayProps) {
  const [viewPly, setViewPly] = useState<number | null>(null);
  const [flipped, setFlipped] = useState(false);
  const [pending, setPending] = useState<string | null>(null);
  const [busy, setBusy] = useState(false);
  const [notice, setNotice] = useState<string | null>(null);
  const [confirmResign, setConfirmResign] = useState(false);
  const [tick, setTick] = useState(0);

  const myColor: Side | null = game.mode === "ai"
    ? (game.ai_color === "w" ? "b" : "w")
    : game.white_id === userId ? "w" : game.black_id === userId ? "b" : null;
  const active = game.status === "active";

  const submit = useCallback(async (uci: string, asAi = false): Promise<boolean> => {
    setBusy(true);
    if (!asAi) {
      try {
        const c = new Chess(game.fen);
        c.move({ from: uci.slice(0, 2), to: uci.slice(2, 4), promotion: uci[4] });
        setPending(c.fen());
      } catch { /* server will reject */ }
    }
    try {
      const r = await act({ action: "move", gameId: game.id, uci, expectedRevision: game.revision, ai: asAi });
      apply(r);
      if (r.flagged) setNotice("Time ran out.");
      setNotice((n) => (r.flagged ? n : null));
      return true;
    } catch (e) {
      setNotice(errMsg(e));
      await sync();
      return false;
    } finally {
      setPending(null);
      setBusy(false);
    }
  }, [game.id, game.fen, game.revision, apply, sync]);

  const ai = useAiPlayer(game, (uci) => submit(uci, true));

  async function run(action: string) {
    setBusy(true);
    try { apply(await act({ action, gameId: game.id })); setNotice(null); }
    catch (e) { setNotice(errMsg(e)); await sync(); }
    finally { setBusy(false); setConfirmResign(false); }
  }

  // Claim the flag when the visible clock reaches zero (any client may; the server verifies).
  useEffect(() => {
    if (!active || game.ply < 2) return;
    const left = remainingAt(game, serverNow())[game.turn];
    if (left > 0) { const t = setTimeout(() => setTick((x) => x + 1), left + 50); return () => clearTimeout(t); }
    let alive = true, tries = 0;
    const attempt = async () => {
      try { const r = await act({ action: "timeout", gameId: game.id }); if (alive) apply(r); }
      catch (e) { if (alive && e instanceof ApiError && e.code === "not_flagged" && tries++ < 6) setTimeout(attempt, 400); }
    };
    attempt();
    return () => { alive = false; };
  }, [game.revision, game.status, tick]);

  const select = useCallback((p: number | null) => setViewPly(p === null || p >= moves.length ? null : Math.max(0, p)), [moves.length]);
  useEffect(() => {
    const onKey = (e: KeyboardEvent) => {
      if ((e.target as HTMLElement)?.closest("input,textarea,select")) return;
      const cur = viewPly ?? moves.length;
      if (e.key === "ArrowLeft") select(Math.max(0, cur - 1));
      else if (e.key === "ArrowRight") select(cur + 1);
      else if (e.key === "Home") select(0);
      else if (e.key === "End") select(null);
    };
    addEventListener("keydown", onKey);
    return () => removeEventListener("keydown", onKey);
  }, [viewPly, moves.length, select]);

  const atLive = viewPly === null;
  const fen = !atLive ? (viewPly === 0 ? START_FEN : moves[viewPly! - 1]?.fen ?? game.fen) : pending ?? game.fen;
  const idx = atLive ? moves.length : viewPly!;
  const lm = idx > 0 ? moves[idx - 1] : undefined;
  const lastMove = lm ? { from: lm.uci.slice(0, 2), to: lm.uci.slice(2, 4) } : null;
  const interactive = atLive && active && !!myColor && game.turn === myColor && !busy && online;
  const orientation: "white" | "black" = (myColor === "b") !== flipped ? "black" : "white";
  const top: Side = orientation === "white" ? "b" : "w";

  const claim = useMemo(() => {
    if (!active || moves.length !== game.ply) return null;
    try { const r = replay(moves.map((m) => m.uci)); return claimableDraw(r.chess, r.keys); } catch { return null; }
  }, [active, game.ply, moves]);

  const offerByOpp = !!game.draw_offer_by && game.draw_offer_by !== userId;

          <button className="btn" onClick={copyPgn}><Copy size={16} /> PGN</button>

            </>
          )}
        </Card>
        {offerByOpp && <p className="text-sm font-medium text-moss-dark">Your opponent offered a draw.</p>}
      </div>

function PlayerBar({ game, color, you }: { game: Game; color: Side; you: boolean }) {
  const [now, setNow] = useState(() => serverNow());
  const running = game.status === "active" && game.ply >= 2 && game.turn === color;
  useEffect(() => {
    setNow(serverNow());
    if (!running) return;
    const t = setInterval(() => setNow(serverNow()), 100);
    return () => clearInterval(t);
  }, [running, game.revision]);

  const ms = remainingAt(game, now)[color];
  const name = (color === "w" ? game.white_name : game.black_name) ?? (color === "w" ? "White" : "Black");
  const avatar = color === "w" ? game.white_avatar : game.black_avatar;

  const toMove = game.status === "active" && game.turn === color;
```

```tsx
function Lobby({ userId }: { userId: string }) {

  const [err, setErr] = useState<string | null>(null);
  const [busy, setBusy] = useState(false);
  const [secs, setSecs] = useState(0);
  const [active, setActive] = useState<Resume[]>([]);
  const waitingRef = useRef<Game | null>(null);
  waitingRef.current = waiting;

  useEffect(() => {
    supabase.from("games").select("id,tc,white_id,white_name,black_name")
      .eq("mode", "online").eq("status", "active").order("updated_at", { ascending: false }).limit(3)
      .then(({ data }) => setActive((data as Resume[]) ?? []));
  }, []);

  async function find() {
    setBusy(true); setErr(null);
    try {
      const r = await act({ action: "queue", tc });
      if (r.game.status === "active") navigate(`/online/${r.game.id}`); else setWaiting(r.game);
    } catch (e) { setErr(errMsg(e)); } finally { setBusy(false); }
  }
  async function cancel() {
    const w = waiting; setWaiting(null);
    if (!w) return;
    try { const r = await act({ action: "cancel_queue", gameId: w.id }); if (r.game.status === "active") navigate(`/online/${r.game.id}`); } catch { /* ignore */ }
  }

  useEffect(() => {
    if (!waiting) return;
    const id = waiting.id, wtc = waiting.tc, t0 = Date.now();
    setSecs(0);

      .on("postgres_changes", { event: "UPDATE", schema: "public", table: "games", filter: `id=eq.${id}` },
        (p) => { if ((p.new as Game).status === "active") navigate(`/online/${id}`); })
      .subscribe();
    const poll = setInterval(async () => {
      try { const r = await act({ action: "sync", gameId: id }); if (r.game.status === "active") navigate(`/online/${id}`); } catch { /* retry */ }
    }, 6000);
    const beat = setInterval(() => { act({ action: "touch_queue", gameId: id }).catch(() => {}); }, 45_000);
    void wtc;
    return () => { clearInterval(tick); clearInterval(poll); clearInterval(beat); supabase.removeChannel(ch); };
  }, [waiting?.id]);

  useEffect(() => () => {
    const w = waitingRef.current;
    if (w) act({ action: "cancel_queue", gameId: w.id }).catch(() => {});
  }, []);

    <div className="mx-auto max-w-xl space-y-4">
```

```ts
    expect(automaticOutcome(new Chess("8/8/8/4k3/8/8/4K3/4N3 w - - 0 1"))?.termination).toBe("insufficient_material");
  });
  it("threefold repetition is claimable, not automatic", () => {
    const o = out([...shuffle, ...shuffle]);
    expect(o.auto).toBeNull();
    expect(o.claim?.termination).toBe("threefold_repetition");
  });
  it("fivefold repetition ends the game automatically", () => {
    expect(out(Array(4).fill(shuffle).flat()).auto?.termination).toBe("fivefold_repetition");
  });
  it("50-move rule is claimable; 75-move rule is automatic", () => {
    const c50 = new Chess("8/8/8/4k3/8/8/4K3/7R w - - 100 80");
    expect(automaticOutcome(c50)).toBeNull();
    expect(claimableDraw(c50)?.termination).toBe("fifty_move");
    expect(automaticOutcome(new Chess("8/8/8/4k3/8/8/4K3/7R w - - 150 100"))?.termination).toBe("seventyfive_move");
  });
  it("checkmate takes precedence over the 75-move rule", () => {
    expect(automaticOutcome(new Chess("R5k1/5ppp/8/8/8/8/8/6K1 b - - 150 90"))?.termination).toBe("checkmate");
  });
});

describe("move validation", () => {
  it("rejects illegal and malformed moves", () => {
    expect(() => applyUci(new Chess(), "e2e5")).toThrow();
    expect(() => applyUci(new Chess(), "e2")).toThrow();
    expect(() => applyUci(new Chess(), "e7e5")).toThrow(); // not black's turn
  });
  it("supports en passant and castling", () => {
    const c = replay(["e2e4", "a7a6", "e4e5", "d7d5"]).chess;
    expect(applyUci(c, "e5d6").flags).toContain("e");
    const k = new Chess("r3k2r/8/8/8/8/8/8/R3K2R w KQkq - 0 1");
    expect(applyUci(k, "e1g1").san).toBe("O-O");
  });
  it("requires an explicit promotion piece and offers all four", () => {
    const fen = "8/P6k/8/8/8/8/8/K7 w - - 0 1";
    expect(() => applyUci(new Chess(fen), "a7a8")).toThrow();
    expect(applyUci(new Chess(fen), "a7a8n").promotion).toBe("n");
    expect(new Chess(fen).moves({ square: "a7", verbose: true }).filter((m) => m.promotion)).toHaveLength(4);
  });
});

describe("timeout and mating material", () => {
  it("timeout is a draw when the opponent cannot mate", () => {
    const c = new Chess("8/8/8/4k3/8/8/4K3/4N3 b - - 0 1");
    expect(timeoutOutcome(c, "b")).toEqual({ result: "1/2-1/2", termination: "timeout" });
  });
  it("timeout is a win when the opponent has mating material", () => {
    const c = new Chess("8/8/8/4k3/8/8/4K3/4R3 b - - 0 1");
    expect(timeoutOutcome(c, "b")).toEqual({ result: "1-0", termination: "timeout" });
  });
  it("evaluates mating material", () => {
    expect(hasMatingMaterial(new Chess("8/8/8/4k3/8/8/4K3/4B3 w - - 0 1"), "w")).toBe(false);
    expect(hasMatingMaterial(new Chess("8/8/8/4k3/8/8/4K3/2B1B3 w - - 0 1"), "w")).toBe(false);
    expect(hasMatingMaterial(new Chess("8/8/8/4k3/8/8/4K3/3NN3 w - - 0 1"), "w")).toBe(true);
  });
});

describe("clocks", () => {
  const g = {
    white_ms: 60_000, black_ms: 60_000, increment_ms: 2_000, turn: "w" as const,
    turn_started_at: new Date(1_000_000).toISOString(), ply: 2, status: "active",
  };
  it("deducts elapsed time and adds the increment after a valid move", () => {
    expect(clockAfterMove(g, 1_010_000)).toEqual({ white_ms: 52_000, black_ms: 60_000, flagged: false });

  it("flags when elapsed time reaches the remaining time", () => {
    expect(clockAfterMove(g, 1_060_000)).toMatchObject({ white_ms: 0, flagged: true });

  it("does not run (or add increment) before both sides have moved", () => {

  });
  it("shows a countdown only for the side to move", () => {
    expect(remainingAt(g, 1_005_000)).toEqual({ w: 55_000, b: 60_000 });
    expect(remainingAt({ ...g, status: "finished" }, 1_005_000)).toEqual({ w: 60_000, b: 60_000 });
  });
  it("offers the six required time controls", () => {
    expect(TIME_CONTROLS.map((t) => t.id)).toEqual(["1+0", "2+1", "3+2", "5+0", "10+0", "10+5"]);

});
```

### `README.md`

````md
# Chess

Play online (Supabase matchmaking + Realtime), play Stockfish (WASM in a Web Worker), and replay your games.
Stack: React + TypeScript + Vite, Tailwind 4, chess.js, react-chessboard, Supabase (Auth, Postgres, Realtime, one Edge Function).

## 1. Create the project (empty repo)
```bash
npm create vite@latest . -- --template react-ts
npm install
npm install @supabase/supabase-js chess.js react-chessboard stockfish lucide-react
npm install -D tailwindcss @tailwindcss/vite vitest
```
Overwrite/add the files from this answer, delete the template leftovers (`src/App.css`, `src/assets`, `public/vite.svg`,
`tsconfig.app.json`, `tsconfig.node.json`, `eslint.config.js`) and replace only the `scripts` block in `package.json`.

```bash
cp .env.example .env.local
npm run dev            # copies Stockfish into public/engine, then starts Vite on http://localhost:5173
npm run typecheck && npm test
npm run build          # output: dist
```

## 2. Database
Open Supabase → SQL Editor → paste all of `supabase/migrations/0001_chess.sql` → Run. It is idempotent, adds the
two tables to the existing `supabase_realtime` publication (it never drops or recreates it), and enables RLS.

## 3. Edge Function `game-action`
```bash
npx supabase login
npx supabase functions deploy game-action --project-ref fhlpgtkftfhqcgppfnud --no-verify-jwt
```
- JWT verification is done inside the function (`auth.getUser`) so it works with the new API-key system; hence `--no-verify-jwt`.
- Required secrets: **none**. `SUPABASE_URL` and `SUPABASE_SERVICE_ROLE_KEY` are injected automatically in hosted functions.
  Only if your project has the legacy service-role key disabled: `npx supabase secrets set SERVICE_KEY=<your secret key>` (server-side only).
- Never put a secret/service-role key in any `VITE_` variable.

## 4. Auth redirect URLs (Supabase → Authentication → URL Configuration → Redirect URLs)
Add only these (the Google provider itself is untouched):
- `http://localhost:5173/**`
- `https://<your-vercel-domain>/**` (add your production domain; also `https://*-<team>.vercel.app/**` if you want preview deployments)

## 5. Vercel
Framework preset: Vite. Build command `npm run build`, output directory `dist`. Environment variables (Production + Preview):
`VITE_SUPABASE_URL`, `VITE_SUPABASE_PUBLISHABLE_KEY`. `vercel.json` handles SPA routing.

## Notes
- **Engine**: Stockfish.js lite single-threaded WASM, copied from `node_modules/stockfish` by `scripts/copy-stockfish.mjs`, which also writes
  `public/engine/manifest.json`. The engine picker lists only what that manifest contains; the version shown is the installed one.
  No cross-origin-isolation headers are needed.
- **Licence**: Stockfish and Stockfish.js are GPL-3.0. Because the app distributes them, publish this repository's source under
  GPL-3.0-compatible terms and add the licence text (`curl -o LICENSE https://www.gnu.org/licenses/gpl-3.0.txt`). The footer links to the
  Stockfish and stockfish.js sources. This is not legal advice.
- **Clocks**: server stores remaining ms + `turn_started_at`; clients count down locally. Clocks start after each side's first move
  (no increment on those two moves). The function validates elapsed time on every move and timeout claim.
- **Draws**: threefold and 50-move are claimable (button appears when available); fivefold, 75-move, stalemate and insufficient material are automatic.
  A flag fall is a draw when the opponent cannot possibly mate. Draw offers are online-only.
- **Queue**: a waiting `games` row is the queue entry (unique per user); it is refreshed every 45 s and ignored by matchmaking after 2 min.
````
